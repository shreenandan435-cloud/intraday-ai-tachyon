                              # pinned <2.5: numba 0.66 does not support numpy 2.5 yet

# ── Crypto history harvest (Track 2: offline RL training data) ─────────
aiohttp==3.14.3               # async Binance Vision daily-archive downloader
                              # (ingestion/binance_vision_scraper.py — offline tool,
                              # never imported by the tick path)

# ── Deep RL (Track 2: offline Transformer-PPO training) ────────────────
# OFFLINE ONLY. Imported exclusively by src/tachyon/model/*, which nothing in the trading path
# loads — see that package's docstring. `import torch` alone costs seconds and hundreds of MB of
# RSS, and a CUDA context inside the orchestrator would put a driver-level allocator behind the
# <1ms inference budget.
torch==2.13.0                 # cp314 wheel verified. THIS PIN IS THE CPU BUILD.
                              # PyPI ships CPU-only wheels for Windows. The RTX 5060 in this
                              # machine is Blackwell (sm_120) and needs a CUDA 12.8+ build, which
                              # only the PyTorch index serves:
                              #   pip install torch==2.13.0 --index-url \
                              #       https://download.pytorch.org/whl/cu128
                              # Everything here runs correctly on the CPU build — it is just slow
                              # to train on. Swap it before the first real training run.
onnx==1.22.0                  # export target; the artefact TensorRT consumes
onnxscript==0.7.1             # required by torch.onnx.export(dynamo=True)
onnxruntime==1.28.0           # exercised in tests/unit/test_model.py to prove the exported
                              # graph still computes what the PyTorch module computes

# ── AI Sentinel (CLAUDE.md §5) ─────────────────────────────────────────
# No google-genai: the Sentinel speaks the Generative Language REST API directly over httpx.
# The SDK brings its own transport and retry behaviour we would have to work around, and the
# only call we make is a single-turn generateContent. One client we fully understand beats a
# large one we mostly do.
httpx==0.28.1                 # the only permitted HTTP client (async; `requests` is banned)
                              # — also the SmartAPI REST transport (CLAUDE.md §6.4)
